## 6.2 认证与会话管理

### 6.2.1 SessionManager

```rust
// security/session.rs
use std::collections::HashMap;
use std::sync::RwLock;
use uuid::Uuid;

pub struct Session {
    pub id: String,
    pub user_id: String,
    pub role: UserRole,
    pub window_labels: Vec<String>,   // 该会话关联的窗口
    pub created_at: DateTime<Utc>,
    pub expires_at: DateTime<Utc>,
}

pub struct SessionManager {
    sessions: RwLock<HashMap<String, Session>>,       // session_id → Session
    window_map: RwLock<HashMap<String, String>>,      // window_label → session_id
}

impl SessionManager {
    /// 创建新会话
    pub fn create_session(&self, user_id: &str, role: UserRole) -> Session {
        let session = Session {
            id: Uuid::new_v4().to_string(),
            user_id: user_id.to_string(),
            role,
            window_labels: vec![],
            created_at: Utc::now(),
            expires_at: Utc::now() + Duration::hours(24),
        };
        self.sessions.write().unwrap()
            .insert(session.id.clone(), session.clone());
        session
    }

    /// 将窗口与会话关联
    pub fn bind_window(&self, session_id: &str, window_label: &str) -> Result<(), AppError> {
        let mut sessions = self.sessions.write().unwrap();
        let mut window_map = self.window_map.write().unwrap();

        let session = sessions.get_mut(session_id)
            .ok_or(AppError::SessionNotFound)?;

        if Utc::now() > session.expires_at {
            return Err(AppError::SessionExpired);
        }

        session.window_labels.push(window_label.to_string());
        window_map.insert(window_label.to_string(), session_id.to_string());
        Ok(())
    }

    /// 根据窗口标签解析会话
    pub fn resolve_from_window(&self, window_label: &str) -> Option<Session> {
        let window_map = self.window_map.read().unwrap();
        let session_id = window_map.get(window_label)?;
        let sessions = self.sessions.read().unwrap();
        sessions.get(session_id).cloned()
    }

    /// 窗口关闭时解绑
    pub fn unbind_window(&self, window_label: &str) {
        let mut window_map = self.window_map.write().unwrap();
        if let Some(session_id) = window_map.remove(window_label) {
            let mut sessions = self.sessions.write().unwrap();
            if let Some(session) = sessions.get_mut(&session_id) {
                session.window_labels.retain(|l| l != window_label);
                // 如果没有关联窗口，清除会话
                if session.window_labels.is_empty() {
                    sessions.remove(&session_id);
                }
            }
        }
    }
}
```

### 6.3.2 权限校验中间件

```rust
// security/auth.rs
pub struct AuthService;

impl AuthService {
    /// 权限校验入口：用于每个 Tauri Command
    pub fn check_permission(
        session: &Session,
        required: Permission,
        resource_owner: Option<&str>,  // 数据行级安全
    ) -> Result<(), AppError> {
        // 1. 检查会话过期
        if Utc::now() > session.expires_at {
            return Err(AppError::SessionExpired);
        }

        // 2. 检查角色权限
        let permissions = session.role.permissions();
        if !permissions.contains(&required) {
            return Err(AppError::Unauthorized(
                format!("Missing permission: {:?}", required)
            ));
        }

        // 3. 数据行级安全检查（非 Admin 只能操作自己的数据）
        if let Some(owner_id) = resource_owner {
            if session.role != UserRole::Admin && owner_id != session.user_id {
                return Err(AppError::Forbidden(
                    "Cannot access resource owned by another user".into()
                ));
            }
        }

        Ok(())
    }
}
```

### 6.3.4 RBAC 数据完整性保护

⚠️ **风险**：RBAC 数据存储在本地 SQLite 中，若用户（或恶意软件）直接修改数据库文件，可篡改 `role` 字段实现提权。

**解决方案**：对权限相关数据增加 HMAC 签名，Rust 侧在读取时验证签名。

#### 签名方案

```
权限数据（user_id, role）
  │
  ▼
HMAC-SHA256（使用数据库加密密钥作为 HMAC Key）
  │
  ▼
签名（32 字节，Base64 编码后存储）
```

#### Rust 侧实现

```rust
// security/rbac_integrity.rs
use hmac::{Hmac, Mac};
use sha2::Sha256;

type HmacSha256 = Hmac<Sha256>;

pub struct RbacIntegrity;

impl RbacIntegrity {
    /// 计算权限数据签名
    pub fn sign(user_id: &str, role: &UserRole, db_key: &[u8]) -> String {
        let mut mac = HmacSha256::new_from_slice(db_key)
            .expect("HMAC can take key of any size");
        
        // 签名输入：user_id + role 字符串
        mac.update(user_id.as_bytes());
        mac.update(role.to_string().as_bytes());
        
        let result = mac.finalize().into_bytes();
        base64::encode(result)
    }
    
    /// 验证签名
    pub fn verify(
        user_id: &str, 
        role: &UserRole, 
        expected_sig: &str,
        db_key: &[u8]
    ) -> bool {
        let expected = Self::sign(user_id, role, db_key);
        expected == expected_sig
    }
}

// 在 UserRepo 读取用户时验证签名
impl UserRepo {
    pub fn find_by_id(&self, id: &str) -> Result<User, AppError> {
        let conn = self.pool.get()?;
        let user: User = conn.query_row(
            "SELECT * FROM users WHERE id = ?1",
            params![id],
            |row| Ok(User::from_row(row))
        )?;
        
        // 验证 HMAC 签名
        let db_key = get_db_key()?;  // 从 SecureStore 读取
        if !RbacIntegrity::verify(&user.id, &user.role, &user.role_sig, &db_key) {
            tracing::error!("RBAC integrity check failed for user {}", id);
            return Err(AppError::SecurityViolation(
                "Permission data tampering detected".into()
            ));
        }
        
        Ok(user)
    }
    
    /// 创建/更新用户时计算签名
    pub fn save(&self, user: &User) -> Result<(), AppError> {
        let conn = self.pool.get()?;
        let db_key = get_db_key()?;
        
        let role_sig = RbacIntegrity::sign(&user.id, &user.role, &db_key);
        
        conn.execute(
            "INSERT OR REPLACE INTO users (id, name, role, role_sig) VALUES (?1, ?2, ?3, ?4)",
            params![user.id, user.name, user.role.to_string(), role_sig],
        )?;
        
        Ok(())
    }
}
```

#### 数据库表结构变更

```sql
ALTER TABLE users ADD COLUMN role_sig TEXT;  -- HMAC 签名

-- 首次迁移时，为现有用户计算签名
UPDATE users SET role_sig = calculate_hmac(user_id || role, db_key);
```

> **安全要点**：
> 1. HMAC Key 使用数据库加密密钥（通过 `get_db_key()` 获取）
> 2. 签名在每次权限变更时重新计算
> 3. 若签名验证失败，拒绝该用户的所有请求（返回 `SecurityViolation` 错误）
> 4. 此方案可检测离线文件篡改，但无法防止用户直接修改 SQLite 文件后同时修改 `role_sig` 字段（需要密钥）

### 6.3.6 审计日志生命周期管理

> **与埋点日志（12）的分工**：审计日志记录**安全敏感操作**（权限变更、数据删除），保留 90 天，支持归档导出——这是合规需求。**业务埋点**（功能使用、性能、错误）归属于 [12-可观测性与数据埋点.md](./12-可观测性与数据埋点.md)，用于产品迭代分析。两者分表存储，独立清理策略。

审计日志会随使用时间持续增长，需要明确的保留与轮转策略以避免无限膨胀。

#### 保留策略

| 参数 | 默认值 | 说明 |
|------|--------|------|
| `retention_days` | 90 天 | 超过此天数的日志可归档/删除 |
| `max_rows` | 100,000 | 表行数上限，超过时触发清理 |
| `batch_delete_size` | 1,000 | 每次清理任务删除的最大行数 |
| `cleanup_interval` | 24 小时 | 后台清理任务执行间隔 |

#### 清理逻辑

```rust
// security/audit_cleanup.rs
use std::time::Duration;

pub struct AuditLogCleaner {
    pool: Arc<r2d2::Pool<SqliteConnectionManager>>,
    retention_days: u32,
    max_rows: u64,
}

impl AuditLogCleaner {
    pub fn new(pool: Arc<r2d2::Pool<SqliteConnectionManager>>) -> Self {
        Self {
            pool,
            retention_days: 90,
            max_rows: 100_000,
        }
    }

    /// 后台清理任务（应用启动时注册）
    pub fn start_background_cleanup(self, app: AppHandle) {
        tokio::spawn(async move {
            loop {
                tokio::time::sleep(Duration::from_secs(24 * 3600)).await;
                
                if let Err(e) = self.cleanup() {
                    tracing::error!("Audit log cleanup failed: {e}");
                }
            }
        });
    }

    fn cleanup(&self) -> Result<(), AppError> {
        let conn = self.pool.get()?;
        
        // 策略 1：按时间清理（超过 retention_days 天）
        let deleted_by_time = conn.execute(
            "DELETE FROM audit_logs 
             WHERE created_at < datetime('now', ?1) 
             LIMIT ?2",
            params![
                format!("-{} days", self.retention_days),
                self.max_rows / 10, // 每次不超过总量 10%
            ],
        )?;
        
        if deleted_by_time > 0 {
            tracing::info!("Audit cleanup: removed {deleted_by_time} expired logs");
        }

        // 策略 2：按数量清理（总行数超 max_rows 时删最旧）
        let row_count: i64 = conn.query_row(
            "SELECT COUNT(*) FROM audit_logs", [], |row| row.get(0)
        )?;
        
        if row_count as u64 > self.max_rows {
            let overflow = (row_count as u64 - self.max_rows).min(1000);
            conn.execute(
                "DELETE FROM audit_logs 
                 WHERE id IN (
                     SELECT id FROM audit_logs 
                     ORDER BY created_at ASC 
                     LIMIT ?1
                 )",
                params![overflow],
            )?;
            tracing::warn!("Audit cleanup: removed {overflow} rows (count exceeded max)");
        }

        Ok(())
    }
    
    /// 归档模式：将旧日志导出为 JSON 文件后删除
    pub fn archive_and_clean(
        &self,
        archive_dir: &Path,
    ) -> Result<PathBuf, AppError> {
        let conn = self.pool.get()?;
        
        let mut stmt = conn.prepare(
            "SELECT * FROM audit_logs WHERE created_at < datetime('now', ?1)"
        )?;
        
        let logs: Vec<AuditLog> = stmt.query_map(
            params![format!("-{} days", self.retention_days)],
            |row| Ok(AuditLog::from_row(row)),
        )?.filter_map(|r| r.ok()).collect();
        
        if logs.is_empty() {
            return Ok(archive_dir.to_path_buf());
        }
        
        let timestamp = chrono::Utc::now().format("%Y%m%d_%H%M%S");
        let archive_path = archive_dir.join(format!("audit_archive_{timestamp}.json"));
        let json = serde_json::to_string_pretty(&logs)?;
        std::fs::write(&archive_path, json)?;
        
        conn.execute(
            "DELETE FROM audit_logs WHERE created_at < datetime('now', ?1)",
            params![format!("-{} days", self.retention_days)],
        )?;
        
        tracing::info!(
            "Audit archive: {} logs archived to {:?}",
            logs.len(), archive_path
        );
        
        Ok(archive_path)
    }
}
```

### 6.4.4 Bridge 回调认证

Python Agent 通过 Bridge 回调读取 SQLite 数据。为防止本地恶意进程伪造 Bridge 请求，需进行认证。

**认证方案**：Rust 启动时生成随机 Token，通过环境变量 `MIRAGEINK_BRIDGE_TOKEN` 传递给 Python。Python 所有 Bridge 请求必须附带 `X-Auth-Token` header。

> **详细实现**：参见 [05-Python-AI扩展架构.md](./05-Python-AI扩展架构.md) 第 5.4.3 节

```rust
// 在 Tauri Command 中验证 Bridge 请求来源
#[tauri::command]
async fn bridge_health_check(auth_token: String, state: State<'_, AppState>) -> Result<String, String> {
    if auth_token != state.bridge_auth_token {
        return Err("Unauthorized".into());
    }
    Ok("ok".into())
}
```

> **安全要点**：
> 1. Token 每次启动随机生成，不写入文件
> 2. Python 进程重启后需重新从环境变量读取 Token
> 3. 若 Token 验证失败，Bridge 返回 401

## 6.5 SQLite 安全

### 6.5.1 数据库加密方案（明确化）

⚠️ **原方案缺陷**：文档之前描述"敏感字段加密存储"，但存在以下问题：
- 部分敏感数据可能遗漏加密
- 攻击者可直接读取未加密的关联数据（如通过 article_id 关联）
- 索引字段（如 `user_id`）无法加密，泄漏用户关系

**新方案**：采用 **SQLCipher 全库加密**，彻底消除敏感字段判断问题。

#### 加密方案设计

```
用户登录密码（或随机生成的主密钥）
  │
  ▼
PBKDF2-HMAC-SHA256（迭代 256000 次）
  │
  ▼
SQLite 加密密钥（32 字节）
  │
  ▼
SQLCipher 使用此密钥加密整个数据库文件
```

#### Rust 侧实现

```rust
// infrastructure/database.rs
use rusqlite::Connection;
use pbkdf2::pbkdf2_hmac;
use sha2::Sha256;
use rand::rngs::OsRng;
use rand::RngCore;

/// 获取或生成数据库加密密钥
fn get_or_create_db_key(passphrase: Option<&str>) -> Result<Vec<u8>, AppError> {
    let secure_store = SecureStore::new();
    
    // 尝试从 Keychain 读取已保存的密钥
    if let Ok(saved_key) = secure_store.get_secret("db_encryption_key") {
        return Ok(base64::decode(saved_key)?);
    }
    
    // 密钥不存在，生成新密钥
    let mut salt = [0u8; 16];
    OsRng.fill_bytes(&mut salt);
    
    let mut key = [0u8; 32];  // 256-bit key
    
    if let Some(pass) = passphrase {
        // 用户提供了密码：使用 PBKDF2 派生密钥
        pbkdf2_hmac::<Sha256>(pass.as_bytes(), &salt, 256_000, &mut key);
    } else {
        // 用户未设置密码（本地模式）：生成随机密钥
        OsRng.fill_bytes(&mut key);
    }
    
    // 将密钥保存到 Keychain（Base64 编码）
    let key_b64 = base64::encode(&key);
    secure_store.store_secret("db_encryption_key", &key_b64)?;
    
    // 同时保存 salt（用于未来密码验证）
    let salt_b64 = base64::encode(&salt);
    secure_store.store_secret("db_salt", &salt_b64)?;
    
    Ok(key.to_vec())
}

pub fn open_database(path: &Path, passphrase: Option<&str>) -> Result<Connection, AppError> {
    let conn = Connection::open(path)?;
    
    // 获取加密密钥
    let key = get_or_create_db_key(passphrase)?;
    let key_hex = hex::encode(&key);
    
    // 启用 SQLCipher 加密
    conn.execute_batch(&format!(
        "PRAGMA key = \"x'{}'\";",  // SQLCipher 接受十六进制密钥
        key_hex
    ))?;
    
    // 验证密钥正确性（执行一个简单的查询）
    conn.execute_batch("SELECT count(*) FROM sqlite_master;")
        .map_err(|_| AppError::DatabaseEncryptionError("Invalid key or corrupted database".into()))?;
    
    // 安全配置
    conn.execute_batch("
        PRAGMA journal_mode = WAL;
        PRAGMA foreign_keys = ON;
        PRAGMA busy_timeout = 5000;
        PRAGMA secure_delete = ON;
        -- SQLCipher 特定配置
        PRAGMA cipher_page_size = 4096;
        PRAGMA cipher_kdf_iter = 256000;
        PRAGMA cipher_hmac_algorithm = HMAC_SHA256;
    ")?;
    
    Ok(conn)
}
```

#### 数据库创建/迁移时的注意事项

```rust
// 首次创建数据库时，需先设置密钥，再创建表
pub fn initialize_database(path: &Path, passphrase: Option<&str>) -> Result<(), AppError> {
    let conn = Connection::open(path)?;
    
    // 1. 先设置密钥
    let key = get_or_create_db_key(passphrase)?;
    let key_hex = hex::encode(&key);
    conn.execute_batch(&format!("PRAGMA key = \"x'{}'\";", key_hex))?;
    
    // 2. 再创建表（会自动加密）
    conn.execute_batch("
        CREATE TABLE IF NOT EXISTS articles (
            id TEXT PRIMARY KEY,
            title TEXT NOT NULL,
            content TEXT,
            version BIGINT NOT NULL DEFAULT 0,
            user_id TEXT NOT NULL,
            created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        );
        -- ... 其他表
    ")?;
    
    Ok(())
}
```

> **安全要点**：
> 1. SQLCipher 加密**整个数据库文件**（包括表结构、索引、临时文件）
> 2. 密钥通过 PBKDF2 派生，迭代 256000 次（2023 年 OWASP 推荐标准）
> 3. 若用户未设置密码，生成随机密钥并保存到系统 Keychain
> 4. 密钥不写入数据库文件或配置文件
> 5. 即使数据库文件被复制，没有密钥也无法解密
