### 10.2.1 ViewModel 集成测试

```typescript
// __tests__/view-models/articleEditorSlice.test.ts
import { configureStore } from '@reduxjs/toolkit';
import { describe, it, expect, vi } from 'vitest';
import articleEditorReducer, {
  updateContent,
  saveArticle,
  saveSucceeded,
  saveFailed,
} from '@/modules/articles/view-models/articleEditorSlice';

// Mock Tauri invoke
vi.mock('@tauri-apps/api/core', () => ({
  invoke: vi.fn(),
}));

describe('articleEditorSlice', () => {
  function createTestStore() {
    return configureStore({
      reducer: { articleEditor: articleEditorReducer },
      preloadedState: {
        articleEditor: {
          article: { id: '1', content: 'old', version: 1 },
          isDirty: false,
          saveStatus: 'idle',
          snapshot: null,
        },
      },
    });
  }

  it('should update content and mark dirty', () => {
    const store = createTestStore();
    store.dispatch(updateContent('new content'));

    const state = store.getState().articleEditor;
    expect(state.article!.content).toBe('new content');
    expect(state.isDirty).toBe(true);
    expect(state.snapshot).not.toBeNull();
  });

  it('should save snapshot on first edit', () => {
    const store = createTestStore();
    store.dispatch(updateContent('first edit'));

    expect(store.getState().articleEditor.snapshot!.content).toBe('old');
  });

  it('should rollback on save failure', () => {
    const store = createTestStore();
    // 先编辑
    store.dispatch(updateContent('optimistic change'));
    expect(store.getState().articleEditor.article!.content).toBe('optimistic change');

    // 保存失败
    store.dispatch(saveFailed());
    expect(store.getState().articleEditor.article!.content).toBe('old');
    expect(store.getState().articleEditor.saveStatus).toBe('failed');
  });

  it('should clear snapshot on save success', () => {
    const store = createTestStore();
    store.dispatch(updateContent('saved change'));
    store.dispatch(saveSucceeded({
      id: '1', content: 'saved change', version: 2,
    }));

    expect(store.getState().articleEditor.snapshot).toBeNull();
    expect(store.getState().articleEditor.isDirty).toBe(false);
  });

  it('should handle version conflict', async () => {
    const { invoke } = await import('@tauri-apps/api/core');
    vi.mocked(invoke).mockRejectedValueOnce({
      code: 'VERSION_CONFLICT',
      serverArticle: { id: '1', content: 'server version', version: 3 },
    });

    const store = createTestStore();
    store.dispatch(updateContent('my version'));

    await store.dispatch(saveArticle());

    expect(store.getState().articleEditor.saveStatus).toBe('conflict');
  });
});
```

```typescript
// __tests__/components/ArticleEditor.test.tsx
import { render, screen, fireEvent } from '@testing-library/react';
import { Provider } from 'react-redux';
import { configureStore } from '@reduxjs/toolkit';
import { ArticleEditor } from '@/modules/articles/views/ArticleEditor';
import articleEditorReducer from '@/modules/articles/view-models/articleEditorSlice';

function renderWithStore(initialState = {}) {
  const store = configureStore({
    reducer: { articleEditor: articleEditorReducer },
    preloadedState: {
      articleEditor: {
        article: { id: '1', title: 'Test', content: 'Hello World', version: 1 },
        isDirty: false,
        saveStatus: 'idle',
        snapshot: null,
        ...initialState,
      },
    },
  });

  return {
    store,
    ...render(
      <Provider store={store}>
        <ArticleEditor />
      </Provider>
    ),
  };
}

describe('ArticleEditor', () => {
  it('should render article content', () => {
    renderWithStore();
    expect(screen.getByDisplayValue('Hello World')).toBeInTheDocument();
  });

  it('should show saving indicator', () => {
    renderWithStore({ saveStatus: 'saving' });
    expect(screen.getByTestId('save-indicator')).toHaveTextContent('保存中...');
  });

  it('should show conflict dialog', () => {
    renderWithStore({
      saveStatus: 'conflict',
      snapshot: { id: '1', content: 'my version', version: 1 },
    });
    expect(screen.getByText('编辑冲突')).toBeInTheDocument();
  });
});
```

```rust
            version INTEGER NOT NULL DEFAULT 1,
            updated_at TEXT NOT NULL,
            created_at TEXT NOT NULL

        INSERT INTO articles VALUES ('1', 'p1', 'Test', 'content', 'draft', 1,
            datetime('now'), datetime('now'));
    ").unwrap();

#[test]
fn test_update_with_optimistic_lock_success() {
    let conn = setup_test_db();
    let usecase = create_test_usecase(&conn);

    let result = usecase.update_with_optimistic_lock("1", "new content", 1);
    assert!(result.is_ok());

    let updated = result.unwrap();
    assert_eq!(updated.content, "new content");
    assert_eq!(updated.version, 2);
}

#[test]
fn test_update_with_optimistic_lock_conflict() {
    let conn = setup_test_db();
    let usecase = create_test_usecase(&conn);

    // 用过期版本号更新
    let result = usecase.update_with_optimistic_lock("1", "new content", 0);
    assert!(result.is_err());
    match result.unwrap_err() {
        AppError::VersionConflict { current_version, client_version } => {
            assert_eq!(current_version, 1);
            assert_eq!(client_version, 0);
        }
        _ => panic!("Expected VersionConflict"),
    }
}
```

### 10.4.1 WebDriver 方案

```typescript
// e2e/article-flow.spec.ts
import { remote } from 'webdriverio';
import { tauriCapabilities } from '@tauri-apps/webdriver';

describe('Article CRUD E2E', () => {
  let client: WebdriverIO.Browser;

  beforeAll(async () => {
    client = await remote(tauriCapabilities({
      application: './src-tauri/target/debug/mirageink',
    }));
  });

  afterAll(async () => {
    await client.deleteSession();
  });

  it('should create and edit an article', async () => {
    // 创建文章
    const createBtn = await client.$('[data-testid="create-article"]');
    await createBtn.click();

    // 输入标题
    const titleInput = await client.$('[data-testid="article-title-input"]');
    await titleInput.setValue('E2E Test Article');

    // 输入内容
    const contentEditor = await client.$('[data-testid="article-content-editor"]');
    await contentEditor.setValue('Content from E2E test');

    // 保存
    const saveBtn = await client.$('[data-testid="save-article"]');
    await saveBtn.click();

    // 验证保存成功
    const status = await client.$('[data-testid="save-status"]');
    await expect(status).toHaveText('已保存');
  });

  it('should handle multi-window sync', async () => {
    // 打开第二个窗口
    await client.executeScript(`
      window.__TAURI_INTERNALS__.invoke('open_editor_window', { articleId: '1' })
    `);

    // 在主窗口编辑
    const contentEditor = await client.$('[data-testid="article-content-editor"]');
    await contentEditor.setValue('Updated from main window');

    // 切换到第二窗口验证
    const windows = await client.getWindowHandles();
    await client.switchToWindow(windows[1]);

    const otherContent = await client.$('[data-testid="article-content-editor"]');
    await expect(otherContent).toHaveValue('Updated from main window');
  });
});
```

测试在 CI 流水线中的位置和组织方式详见 [11 — CI/CD 持续集成与交付](./11-CICD%E6%8C%81%E7%BB%AD%E9%9B%86%E6%88%90%E4%B8%8E%E4%BA%A4%E4%BB%98.md)。本节给出测试在其中的最小摘要：

```yaml
# CI 中三个语言栈的测试并行执行
jobs:
  test-rust:      # cargo test --all-features + tarpaulin 覆盖率
  test-frontend:  # pnpm test --coverage
  test-python:    # pytest --cov
  e2e-tests:      # macOS runner, Tauri WebDriver（仅 PR 含关键路径变更时触发）
```

CI 构建验证矩阵覆盖 macOS (ARM + x86_64)、Windows、Linux 三平台。安全审计（cargo-audit / npm audit / pip-audit / CodeQL）在每次 PR 和每周定时执行。
