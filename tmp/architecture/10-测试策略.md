# 10 — 测试策略

## 10.1 测试金字塔

```
        ┌──────┐
        │ E2E   │  ≅15%   Tauri WebDriver / 多窗口场景
       ┌┴──────┴┐
       │ 集成测试│ ≅20%  vitest + tauri::test + Python
      ┌┴────────┴┐
      │ 组件测试  │ ≅25%  vitest + React Testing Library
     ┌┴──────────┴┐
     │  单元测试   │ ≅40%  #[test] + vitest + pytest
    └─────────────┘
```

> **E2E 覆盖率提升理由**（从 5% → 15%）：
> - Tauri 桌面应用的核心特性是多窗口交互，单元测试无法覆盖跨窗口状态同步场景
> - 双窗口同时编辑同一文章是最容易出 bug 的场景（乐观锁冲突 + 事件同步时序）
> - E2E 占比从 5% 提升至 15% 是经过行业实践验证的桌面应用测试最佳比例

### 10.1.1 测试分层说明

| 层 | 占比 | 工具 | 重点关注 |
|---|---|---|---|
| 单元测试 | 40% | vitest / `#[test]` / pytest | 纯函数、数据转换、业务规则 |
| 组件测试 | 25% | vitest + RTL | 组件渲染、用户交互、状态变化 |
| 集成测试 | 20% | tauri::test / FastAPI TestClient | UseCase、Command、IPC 路径 |
| E2E | 15% | Tauri WebDriver | 多窗口交互、完整用户流程 |

```typescript
// __tests__/view-models/articleEditorSlice.test.ts
import { configureStore } from '@reduxjs/toolkit';
import { describe, it, expect, vi } from 'vitest';
import articleEditorReducer, {
  updateContent,
  saveArticle,
  saveSucceeded,
  saveFailed,
} from '@/modules/articles/view-models/articleEditorSlice';

// Mock Tauri invoke
vi.mock('@tauri-apps/api/core', () => ({
  invoke: vi.fn(),
}));

describe('articleEditorSlice', () => {
  function createTestStore() {
    return configureStore({
      reducer: { articleEditor: articleEditorReducer },
      preloadedState: {
        articleEditor: {
          article: { id: '1', content: 'old', version: 1 },
          isDirty: false,
          saveStatus: 'idle',
          snapshot: null,
        },
      },
    });
  }

  it('should update content and mark dirty', () => {
    const store = createTestStore();
    store.dispatch(updateContent('new content'));

    const state = store.getState().articleEditor;
    expect(state.article!.content).toBe('new content');
    expect(state.isDirty).toBe(true);
    expect(state.snapshot).not.toBeNull();
  });

  it('should save snapshot on first edit', () => {
    const store = createTestStore();
    store.dispatch(updateContent('first edit'));

    expect(store.getState().articleEditor.snapshot!.content).toBe('old');
  });

  it('should rollback on save failure', () => {
    const store = createTestStore();
    // 先编辑
    store.dispatch(updateContent('optimistic change'));
    expect(store.getState().articleEditor.article!.content).toBe('optimistic change');

    // 保存失败
    store.dispatch(saveFailed());
    expect(store.getState().articleEditor.article!.content).toBe('old');
    expect(store.getState().articleEditor.saveStatus).toBe('failed');
  });

  it('should clear snapshot on save success', () => {
    const store = createTestStore();
    store.dispatch(updateContent('saved change'));
    store.dispatch(saveSucceeded({
      id: '1', content: 'saved change', version: 2,
    }));

    expect(store.getState().articleEditor.snapshot).toBeNull();
    expect(store.getState().articleEditor.isDirty).toBe(false);
  });

  it('should handle version conflict', async () => {
    const { invoke } = await import('@tauri-apps/api/core');
    vi.mocked(invoke).mockRejectedValueOnce({
      code: 'VERSION_CONFLICT',
      serverArticle: { id: '1', content: 'server version', version: 3 },
    });

    const store = createTestStore();
    store.dispatch(updateContent('my version'));

    await store.dispatch(saveArticle());

    expect(store.getState().articleEditor.saveStatus).toBe('conflict');
  });
});
```

## 10.4 E2E 测试

### 10.4.1 端到端测试场景矩阵

由于 Tauri 桌面应用的核心特征是多窗口交互，E2E 测试必须重点覆盖以下场景：

| 场景编号 | 场景描述 | 严重性 | 验证点 |
|---|---|---|---|
| **E2E-01** | **双窗口同时编辑同一文章** | CRITICAL | 乐观锁冲突检测 + UI 提示 |
| **E2E-02** | 主窗口编辑 → 大纲窗口同步刷新 | HIGH | 事件广播 + Store 更新 |
| **E2E-03** | 窗口 A 删除文章 → 窗口 B 自动关闭编辑 | HIGH | 数据一致性 + 窗口生命周期 |
| **E2E-04** | 离线编辑 3 条 → 上线后同步 | HIGH | 离线队列 + 同步引擎 |
| **E2E-05** | Python Agent 崩溃重启 → 前端自动恢复 | MEDIUM | 错误恢复 + 状态同步 |
| **E2E-06** | 快速连续保存 10 次 → 版本号正确递增 | MEDIUM | 并发写入 + 乐观锁 |
| **E2E-07** | 主题切换同步到所有窗口 | LOW | 配置同步 |

### 10.4.3 关键场景：双窗口冲突编辑（E2E-01）

这是整个应用中最复杂的交互场景，必须在 E2E 测试中完整覆盖。

```typescript
// e2e/multi-window-conflict.spec.ts
import { remote } from 'webdriverio';
import { tauriCapabilities } from '@tauri-apps/webdriver';

describe('Multi-Window Conflict Resolution (E2E-01)', () => {
  let client: WebdriverIO.Browser;
  let windowA: string, windowB: string;

  beforeAll(async () => {
    client = await remote(tauriCapabilities({
      application: './src-tauri/target/debug/mirageink',
    }));
  });

  afterAll(async () => {
    // 关闭所有窗口
    const handles = await client.getWindowHandles();
    for (const handle of handles) {
      await client.switchToWindow(handle);
      await client.closeWindow();
    }
    await client.deleteSession();
  });

  beforeEach(async () => {
    // 1. 在主窗口中创建测试文章
    windowA = (await client.getWindowHandles())[0];
    await client.switchToWindow(windowA);
    await client.$('[data-testid="create-article"]').click();
    await client.$('[data-testid="article-title-input"]').setValue('Conflict Test');
    await client.$('[data-testid="article-content-editor"]').setValue('Original Content');
    await client.$('[data-testid="save-article"]').click();
    await client.pause(1000); // 等待保存完成

    // 2. 打开第二个编辑器窗口（同一篇文章）
    await client.executeScript(`
      window.__TAURI_INTERNALS__.invoke('open_editor_window', { 
        articleId: 'conflict-test-1' 
      })
    `);
    await client.pause(500);
    
    const handles = await client.getWindowHandles();
    windowB = handles[1];
  });

  it('should detect version conflict when both windows edit the same article', async () => {
    // Step 1: 在窗口 A 中编辑
    await client.switchToWindow(windowA);
    const editorA = await client.$('[data-testid="article-content-editor"]');
    await editorA.clearValue();
    await editorA.setValue('Content edited in Window A');
    await client.$('[data-testid="save-article"]').click();
    await client.pause(500);

    // Step 2: 在窗口 B 中编辑（版本号已过时）
    await client.switchToWindow(windowB);
    const editorB = await client.$('[data-testid="article-content-editor"]');
    await editorB.clearValue();
    await editorB.setValue('Content edited in Window B');
    await client.$('[data-testid="save-article"]').click();
    await client.pause(500);

    // Step 3: 验证冲突检测
    // 窗口 B 应该显示冲突对话框
    const conflictDialog = await client.$('[data-testid="conflict-dialog"]');
    await expect(conflictDialog).toBeDisplayed();
    
    // 冲突对话应包含两个版本
    const localVersion = await client.$('[data-testid="conflict-local"]');
    const serverVersion = await client.$('[data-testid="conflict-server"]');
    await expect(localVersion).toHaveText('Content edited in Window B');
    await expect(serverVersion).toHaveText('Content edited in Window A');

    // Step 4: 解决冲突 - 用户选择保留本地版本
    await client.$('[data-testid="resolve-local"]').click();
    await client.pause(500);

    // Step 5: 验证冲突解决结果
    // 窗口 B 应显示本地版本
    const finalContent = await client.$('[data-testid="article-content-editor"]');
    await expect(finalContent).toHaveValue('Content edited in Window B');

    // 窗口 A 应通过事件同步获取最新版本
    await client.switchToWindow(windowA);
    await client.pause(300);
    const syncedContent = await client.$('[data-testid="article-content-editor"]');
    await expect(syncedContent).toHaveValue('Content edited in Window B');
  });

  it('should automatically close editor window when article is deleted (E2E-03)', async () => {
    // 窗口 A 删除文章
    await client.switchToWindow(windowA);
    await client.$('[data-testid="delete-article"]').click();
    await client.$('[data-testid="confirm-delete"]').click();
    await client.pause(1000);

    // 窗口 B 应自动关闭（或显示"文章已删除"提示）
    await client.switchToWindow(windowB);
    const deletedNotice = await client.$('[data-testid="article-deleted-notice"]');
    await expect(deletedNotice).toBeDisplayed();
  });

  it('should handle rapid consecutive saves without version conflicts (E2E-06)', async () => {
    await client.switchToWindow(windowA);
    const editor = await client.$('[data-testid="article-content-editor"]');
    
    // 连续快速保存 10 次
    for (let i = 1; i <= 10; i++) {
      await editor.setValue(`Save iteration ${i}`);
      await client.$('[data-testid="save-article"]').click();
      await client.pause(100);
    }

    // 验证最终版本号正确
    const versionDisplay = await client.$('[data-testid="article-version"]');
    await expect(versionDisplay).toHaveText('11'); // 初始 1 + 10 次保存
  });
});
```
